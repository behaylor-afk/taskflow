// ============================================================ // TaskFlow — Google Apps Script Backend v6 (subtask support) // Deploy as Web App: Execute as Me, Anyone can access // ============================================================ var SHEET_TASKS = "Tasks"; var SHEET_APPTS = "Appointments"; function getOrCreate(name, headers) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); sh.setFrozenRows(1); } return sh; } function tasksSheet() { return getOrCreate(SHEET_TASKS, ["id","name","entity","startDate","dueDate","recurrence","completedDates","subtasks","permanentlyDone"]); } function apptsSheet() { return getOrCreate(SHEET_APPTS, ["date","appointments"]); } var SHEET_HABITS = "Habits"; function habitsSheet() { return getOrCreate(SHEET_HABITS, ["data"]); } function getHabits() { var sh = habitsSheet(); var rows = sh.getDataRange().getValues(); if (rows.length <= 1 || !rows[1][0]) return []; try { return JSON.parse(String(rows[1][0])); } catch(e) { return []; } } function saveHabits(habits) { var sh = habitsSheet(); var val = JSON.stringify(habits || []); if (sh.getLastRow() <= 1) sh.appendRow([val]); else sh.getRange(2,1).setValue(val); return {ok:true}; } function makeResponse(data, callback) { var json = JSON.stringify(data); if (callback) { return ContentService .createTextOutput(callback + "(" + json + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getTasks() { var rows = tasksSheet().getDataRange().getValues(); if (rows.length <= 1) return []; return rows.slice(1).map(function(r) { var subtasks = []; if (r[7]) { try { subtasks = JSON.parse(String(r[7])); } catch(e) { subtasks = []; } } return { id: String(r[0]), name: String(r[1]), entity: String(r[2]), startDate: String(r[3]), dueDate: String(r[4]), recurrence: String(r[5]), completedDates: r[6] ? String(r[6]).split(",").filter(Boolean) : [], subtasks: subtasks, permanentlyDone: r[8] === true || String(r[8]).toLowerCase() === "true" }; }); } function getAppts() { var rows = apptsSheet().getDataRange().getValues(); var out = {}; rows.slice(1).forEach(function(r) { if (r[0]) out[String(r[0])] = r[1] ? String(r[1]).split("||").filter(Boolean) : []; }); return out; } function saveTask(task) { var sh = tasksSheet(), rows = sh.getDataRange().getValues(); var cd = (task.completedDates || []).join(","); var subtasksJson = JSON.stringify(task.subtasks || []); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(task.id)) { sh.getRange(i+1,1,1,9).setValues([[ task.id, task.name, task.entity, task.startDate, task.dueDate, task.recurrence, cd, subtasksJson, !!task.permanentlyDone ]]); return {ok:true, action:"updated"}; } } sh.appendRow([task.id, task.name, task.entity, task.startDate, task.dueDate, task.recurrence, cd, subtasksJson, !!task.permanentlyDone]); return {ok:true, action:"inserted"}; } function deleteTask(id) { var sh = tasksSheet(), rows = sh.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(id)) { sh.deleteRow(i+1); return {ok:true}; } } return {ok:false, error:"Not found"}; } function saveAppts(date, appts) { var sh = apptsSheet(), rows = sh.getDataRange().getValues(); var val = (appts || []).join("||"); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(date)) { sh.getRange(i+1,1,1,2).setValues([[date,val]]); return {ok:true}; } } sh.appendRow([date,val]); return {ok:true}; } function handleAction(body) { var action = body.action; if (action === "saveTask") return saveTask(body.task); if (action === "deleteTask") return deleteTask(body.id); if (action === "saveAppts") return saveAppts(body.date, body.appts); if (action === "saveHabits") return saveHabits(body.habits); return {ok:false, error:"Unknown action: " + action}; } function doGet(e) { try { var p = (e && e.parameter) ? e.parameter : {}; var callback = p.callback || null; if (p.payload) return makeResponse(handleAction(JSON.parse(p.payload)), callback); return makeResponse({ok:true, tasks:getTasks(), appts:getAppts(), habits:getHabits()}, callback); } catch(err) { return makeResponse({ok:false, error:err.message}); } } function doPost(e) { try { var raw = (e.postData && e.postData.contents) ? e.postData.contents : "{}"; return makeResponse(handleAction(JSON.parse(raw))); } catch(err) { return makeResponse({ok:false, error:err.message}); } }
